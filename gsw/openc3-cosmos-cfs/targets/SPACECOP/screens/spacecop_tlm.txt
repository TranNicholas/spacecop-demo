# Fixed screen size (WIDTH HEIGHT POLL_PERIOD). A fixed size sizes more
# predictably than AUTO when a large TEXTBOX is involved.
SCREEN 1050 850 1.0

VERTICAL
  TITLE "<%= @target_name %> SPACECOP"

  # Housekeeping counters
  HORIZONTAL
    LABELVALUE SPACECOP SPACECOP_HK_TLM CMD_COUNT
    SPACER 20 0
    LABELVALUE SPACECOP SPACECOP_HK_TLM CMD_ERR_COUNT
  END

  SPACER 0 10

  LABEL "Latest IDS report:"
    SETTING RAW font-size 22px
    SETTING TEXTCOLOR white
    SETTING BACKCOLOR 30 30 30
    SETTING WIDTH 100%

  # Stock widget: shows only the latest downlinked IDS report. The supplied
  # scrolling log does not work on COSMOS 7.2.1: "WIDGET IDSLOG" requests a
  # nonexistent WidgetWidget, and IdslogWidget mixes in only the base Widget,
  # so even as "IDSLOG ..." it never subscribes to IDS_MSG.
  TEXTBOX SPACECOP SPACECOP_REPORT_TLM IDS_MSG 1000 700 FORMATTED
    SETTING RAW width 1000px
    SETTING RAW height 750px
END
